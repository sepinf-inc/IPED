###########################################################################
# Image thumbnail extraction configuration
###########################################################################

# Uses external tools to render hundreds of image formats not supported by Java (slower).
# ImageMagick is already distributed for Windows systems.
# For other systems, like Linux, it must be installed and in the system PATH.
enableExternalConv = true

# Choose between "graphicsmagick" and "imagemagick" for external image conversion.
# ImageMagick supports about twice as many image formats, and decoding of common formats was better in a past evaluation.
# Occasionally, GraphicsMagick may decode some formats better.
externalConversionTool = imagemagick

# Density used for external conversion of vector images for low-resolution output (thumbnails).
lowResDensity = 96

# Density used for external conversion of vector images for high-resolution output.
highResDensity = 250

# Maximum number of megapixels (MP) of an image that can reside in the pixel cache memory
# used by the external conversion process. Higher values will consume more memory, but can make
# the conversion process much faster for some images.
maxMPixelsInMemory = 32

# Timeout (seconds) to wait for external image decoding.
imgConvTimeout = 20

# Additional time (seconds) per megabyte to be added to imgConvTimeout above.
imgConvTimeoutPerMB = 2

# Pixel size of generated thumbnails.
imgThumbSize = 256

# Compression level of the generated thumbnail image.
# From 0 to 100: 0 is very high compression and very low quality; 100 is highest quality and minimum compression.
compression = 60

# Extracts embedded thumbnails from EXIF metadata. Although this method is faster than regenerating them,
# embedded thumbnails usually have lower quality and may differ from the original (full-size) image.
extractThumb = false

# Number of threads for on-demand thumbnail creation in the gallery. "default" uses the number of CPU cores.
# Decrease this value if you have high memory usage problems.
galleryThreads = default

# Logs the rendering of each image in the gallery. Could generate huge logs or
# slow down gallery rendering depending on the log location.
logGalleryRendering = false

# Creates view images (to speed up rendering) for the following MIME types.
# Used only for images that require external conversion. Use ";" as a separator.
mimesToCreateView = image/heic; image/heif

# Maximum size (in pixels) of the generated view image.
maxViewImageSize = 2400
