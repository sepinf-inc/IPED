# Added unallocated space will be indexed. "addUnallocated" (FileSystemConfig.txt) and "parseUnknownFiles" (ParsingTaskConfig.txt) must be enabled.
indexUnallocated = false

# Converts text to lowercase before indexing, making searches case-insensitive.
# Disable only in exceptional cases to generate better dictionaries for case-sensitive password cracking.
convertCharsToLowerCase = true

# Removes diacritics and converts characters to their equivalent ASCII characters.
convertCharsToAscii = true

# Defines additional characters to be indexed besides letters and numbers, i.e., these characters will no longer be treated as separators.
extraCharsToIndex =

# Do not index words with characters outside the Latin-1 charset, such as Chinese, Cyrillic, Arabic...
# Default is false.
filterNonLatinChars = false

# Size (bytes) of the text segments extracted from items before indexing. Includes all items,
# not just the ones indexed via strings. This avoids OutOfMemory errors while indexing items with large chunks of extracted text. 
textSplitSize = 10485760

# Uses NIOFSDirectory instead of MMAPDirectory to open the index (https://lucene.apache.org/core/4_9_0/core/org/apache/lucene/store/FSDirectory.html).
# It is a bit slower, but prevents JVM crashes when reading the index over network shares.
useNIOFSDirectory = false

# Forces index merging into a single segment, reducing its size and optimizing searches from optical media.
# This merging is costly and unnecessary if the index is accessed from a hard disk.
forceMerge = false

# Stores term vectors in the index, needed to search for similar documents.
# Increases index size by up to 30%.
storeTermVectors = true

# Maximum word length to be indexed.
maxTokenLength = 255

# Interval to commit partial processing results, so processing can be resumed later if stopped.
# Partial commits can be a very costly operation; be careful if you change the default.
commitIntervalSeconds = 1800
