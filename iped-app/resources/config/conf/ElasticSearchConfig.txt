# ElasticSearch Task configuration

# Hostname or IP address of the Elastic cluster
host = Sepinf-hpw03

# Port on which Elastic is listening
port = 9200

# Protocol to use: http or https
protocol = http

# Number of primary index shards (sub-indexes) into which each index is split.
# Often each index shard goes to a different cluster node. 
index.number_of_shards = 1

# Number of replicas for each index shard. Often goes to a different node.
index.number_of_replicas = 1

# Index lifecycle policy. You must define it in Elastic. For OpenDistro, it must be commented out.
# index.lifecycle.name = default_policy

# Minimum size of bulk index requests in bytes. Items are accumulated until this size is reached.
# This is a per-thread setting.
min_bulk_size = 8388608

# Minimum number of items in bulk index requests. Actually, an OR condition is applied with min_bulk_size, not an AND.
# This is a per-thread setting.
min_bulk_items = 1000

# Maximum number of simultaneous async bulk requests. This is a per-thread setting.
# Heap usage on the client is about num_threads x min_bulk_size x max_async_requests.
max_async_requests = 5

# Timeout for index requests
timeout_millis = 3600000

# Timeout for connecting to Elastic
connect_timeout_millis = 5000

# Maximum number of indexed fields
index.mapping.total_fields.limit = 10000

# Uses a custom analyzer that tokenizes on non-digits and non-Latin-1 characters,
# converts them to lowercase, and uses an ASCII folding filter.
useCustomAnalyzer = true

# Validates the SSL certificate; it is not recommended to disable this in production.
validateSSL = true

# Indexes the extracted text using term vectors. This improves highlighting speed, but increases the index size by around 30%.
useTermVector = false

# Number of times the index request will be retried. A value of -1 retries forever.
retries = 1
