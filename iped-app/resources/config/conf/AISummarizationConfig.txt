##############################################################################
# Configuration file for AI-based summarization of evidence items.
##############################################################################

# Before enabling enableAISummarization in IPEDConfig.txt, install requests and
# beautifulsoup4 in the Python environment used by IPED. Installing requests also
# installs urllib3 and its other dependencies.
# Windows: run these commands from the IPED installation directory:
#   .\python\python.exe .\python\get-pip.py
#   .\python\python.exe -m pip install requests beautifulsoup4
# Linux: run this command in the Python environment used by IPED:
#   python3 -m pip install requests beautifulsoup4
# For Python/JEP setup, including NumPy requirements, see:
# https://github.com/sepinf-inc/IPED/wiki/User-Manual#python-modules
# AISummarizationTask does not use NumPy directly.

# IP:PORT of the service/central node used by the AISummarizationTask implementation.
# remoteServiceAddress = 127.0.0.1:8000

# Remote service communication parameters used by AISummarizationTask.
busySleepTime = 10.0
nonBusySleepTime = 1.0
maxAttemptsNonBusyRetries = 10
connectionTimeout = 5
requestTimeout = 7200

# Select which chat parsers should feed summarization.
# internal: use IPED internal chat MIME types (currently WhatsApp, Telegram and Threema)
#           and only UFED chat previews without an equivalent internal parser.
# external: use only UFED chat previews.
parsersToUse = internal

# Minimum item content length to perform summarization in characters
minimumContentLength = 1000

# Options for chat analysis with AI LLM.
# When 'enableChatAnalysis' is true, custom questions in 'questions' are sent to the AI summarizer,
# and the returned scores (0-999) will be available as attributes listed in 'questionAttributes'.
enableChatAnalysis = false
