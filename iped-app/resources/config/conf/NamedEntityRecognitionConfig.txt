# org.apache.tika.parser.ner.nltk.NLTKNERecogniser (NLTK)
NERImpl = iped.engine.task.ner.spacy.SpaCyNERecogniser

# Map between language and NER model. Syntax:
# langModel_X = language : model_name_or_path

# For SpaCy:
# Models can be installed via:
#   python -m spacy download pt_core_news_sm
