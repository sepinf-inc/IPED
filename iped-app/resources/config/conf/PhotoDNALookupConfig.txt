#############################
# PhotoDNA Lookup Task configuration
#############################

# Uses only PhotoDNA hashes from HashDB entries with a specific "status" in the lookup.
# Leaving this empty means that all PhotoDNA entries will be loaded.
# Commas can be used to separate multiple statuses, e.g., "pedo, alert".
statusHashDBFilter = pedo

# Maximum distance between PhotoDNA hashes to consider two images similar.
# Used to look up PhotoDNA hashes in a hash database.
# The larger the number, the SLOWER the lookup will be, and you'll get more false positives and fewer false negatives.
# Distances below this threshold will set the 'photoDnaDb:hit' property to 'true' and load other properties from the
# hash database (like 'photoDnaDb:status' and 'photoDnaDb:set', when present).
maxSimilarityDistance = 40000

# Searches for PhotoDNA of rotated (90, 180, and 270 degrees) and flipped images in the PhotoDNA database.
searchRotatedAndFlipped = true

# PhotoDNA minimum entropy (between 0 and 1). Images below this threshold (low information) are not matched
# to avoid false positives. A value of 0 matches all images.
minEntropy = 0.7
