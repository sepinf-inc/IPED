# Face recognition feature based on the Face Recognition project (https://pypi.org/project/face-recognition/)
# You must install Python and some dependencies; see https://github.com/sepinf-inc/IPED/wiki/User-Manual#facerecognition
# If enabled, you can search for faces from the analysis interface (check the options menu).

# Number of external face recognition processes. If not set, the number of CPU logical cores will be used.
# You can set this if you have memory usage problems, e.g., if you are using a GPU with very little memory.
#numFaceRecognitionProcesses = 4

# Max width or height used to resize images before face detection. This dramatically reduces
# face detection time on high-resolution images, although small faces may be missed.
# This can be increased, but processing will take much longer.
maxResolution = 1024

# Face detection algorithm. Possible values: 'hog' or 'cnn'.
# The latter is more accurate, but is much slower. It should be used only with a good GPU.
faceDetectionModel = hog

# Number of times the image will be upsampled before recognition. Default is 1x (doubles the image size).
# This improves detection of small faces, but is limited by the maxResolution setting.
upSampling = 1

# Minimum size (in pixels) to apply face recognition to an image. 
# Images with either dimension (height or width) smaller than this threshold are ignored by this task.   
minSize = 48
