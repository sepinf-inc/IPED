# -----------------------------------------------------------------------------

# Comma-separated list of specific ALEAPP plugins to run.
# Use this to create an inclusion list, forcing IPED to run ONLY the specified plugins.
# If left empty (default behavior), IPED will run all available ALEAPP plugins
# except for those defined in the 'excludePlugins' property below.
# Example: includePlugins=firefox,teams,googleCalendar
includePlugins=

# -----------------------------------------------------------------------------

# Comma-separated list of ALEAPP plugins to specifically exclude from processing.
# This is useful for disabling plugins that are redundant with native IPED parsers,
# are irrelevant to the case, or produce undesired output.
#
# The default exclusions are listed below with justifications:
# - walStrings: Excluded because IPED natively parses strings from unparsed files.
# - WhatsApp: Excluded because IPED uses its own robust, internal parser for WhatsApp artifacts.
excludePlugins=walStrings,WhatsApp

# -----------------------------------------------------------------------------

# What to run over the file system of a UFDR. The data decoded by UFED (calls, contacts, chats, accounts...)
# is already imported by IPED from the report, so decoding the same files again with ALEAPP duplicates it.
# Possible values:
#   all      - run every enabled plugin (results may duplicate the data decoded by UFED).
#   external - prefer the data decoded by UFED: a plugin is skipped when the UFDR holds at least one of the
#              UFED models configured as its equivalent ('skipIfUfedHas.*' properties below). Plugins with no
#              equivalent model, or whose models UFED did not decode in that UFDR, run as usual.
#   none     - do not run ALEAPP over UFDR extractions.
# This has no effect on other kinds of extraction (zip, folder, android backup).
ufdrDecodingToUse = external

# ALEAPP plugins to skip when the UFDR already has the same data decoded by UFED, used when
# 'ufdrDecodingToUse = external'. The key names the ALEAPP plugin, the value the equivalent UFED models:
#   skipIfUfedHas.<module> = <models>              all the artifacts of an ALEAPP module (python file name, as in 'excludePlugins')
#   skipIfUfedHas.<module>.<artifact> = <models>   one artifact of an ALEAPP module (takes precedence over the module one)
# <models> is a comma-separated list of UFED models, the plugin is skipped if the UFDR has any of them:
#   Type          models of that type, from any source (e.g. InstalledApplication)
#   Type:Source   models of that type whose 'Source' is the given application (e.g. Chat:Telegram)
#   Type:-        models of that type with no source (as UFED reports the native call log and contact list)
# Type and Source are the ones shown by IPED in the UFED items ('application/x-ufed-<type>' and 'ufed:Source').
# The skip is by plugin: if a plugin decodes several applications in one run (e.g. every Chromium based
# browser) and UFED decoded just one of them, the others are not decoded by ALEAPP either. Remove the line
# of a plugin to always run it.
# The WhatsApp lines only matter if the WhatsApp module is removed from 'excludePlugins' above.
skipIfUfedHas.calllog.get_calllog = Call:-
