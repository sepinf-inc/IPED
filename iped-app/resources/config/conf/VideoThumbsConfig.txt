###############################################
# Video Thumbnail Extraction Configuration    #
###############################################

# MPlayer for Windows is already distributed alongside this software.
# On Linux, MPlayer must be compiled and installed on the system PATH.

# Thumbnail extraction configuration, format: <maximum frame size in pixels>,<columns>,<rows>. 
# The higher the resolution or number of thumbnails, the slower the extraction.
Layout = 320,4,5

# Compression level of the generated image file.
# From 0 to 100: 0 is very high compression and very low quality; 100 is higher quality and lowest compression.
compression = 60

# Timeouts (in seconds) used by the MPlayer monitor.
# Change with caution, as high values can slow down the processing of corrupted videos.
# Low values can abort thumbnail extraction for valid videos.
#    - First execution timeout (must be higher because MPlayer may update its font cache).
#    - Metadata extraction
#    - Thumbnail extraction
Timeouts = 300,15,45

# Enable or disable logging of MPlayer messages (only for debugging problems).
Verbose = false

# Gallery thumbnail configuration for videos.
# Comment out to disable (the full video preview image will be shown in the gallery).
# Format: <maximum gallery image size in pixels, minimum number of frames, maximum number of frames>.
GalleryThumbs = 320,2,3

# Enables video frame extraction as video subitems.
# If enabled, in parallel with videoThumbs creation, video frames are created
# as video subitems, so they will be able to participate in the 
# image processing pipeline (OCR, image similarity, etc.).
enableVideoThumbsSubitems = false

# If enabled, uses a JavaScript function to compute the number of frames to extract from videos as subitems.
# You can use the 'duration' variable: the video duration in seconds.
# This applies only if 'enableVideoThumbsSubitems' is enabled.
# A minimum of <columns> x <rows> frames from the 'Layout' option will be extracted.
#numFramesEquation = Math.sqrt(duration)

# Enables video frame extraction using the original video resolution for Video Thumbs Subitems.
# If enabled, the first value of the 'Layout' option above will be ignored.
# This should help with face detection in videos and other future features (such as video OCR, etc.).
enableVideoThumbsOriginalDimension = false

# Upper bound for width/height dimensions used when extracting video frames.
# This applies to and takes precedence over all options above.
maxDimensionSize = 1024
