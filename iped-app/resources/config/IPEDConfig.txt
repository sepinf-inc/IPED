########################################################################
# IPED - Processing Settings
########################################################################

# Enables file hash calculation.
enableHash = true

# Enables PhotoDNA hash calculation. If you have the rights to use it (law enforcement fighting child exploitation)
# and have received the library, you must put it into the plugins folder.
enablePhotoDNA = false

# Enables hash lookup in the IPED database.
enableHashDBLookup = false

# Enables PhotoDNA lookup in the IPED hash database.
enablePhotoDNALookup = false

# Enables nudity detection. For each image, it creates the attributes "nudityScore" (1 to 1000)
# and "nudityClass" (1 to 5) for ranking and/or filtering.
enableLedDie = false

# [Experimental] Enables nudity detection using the Yahoo OpenNSFW deep learning algorithm.
# You must install Python and some dependencies to enable it; see https://github.com/sepinf-inc/IPED/wiki/User-Manual#python-modules
# This is more than an order of magnitude slower than the LedDie algorithm on the CPU; using a good GPU is highly recommended.
# It gives higher scores than the LedDie algorithm for images/videos with genitals or explicit sex, mainly because of training dataset differences.
# An attribute "nsfw_nudity_score" from 0 to 100 will be created.
enableYahooNSFWDetection = false

# [Experimental] Detects Child Sexual Abuse Material (CSAM) using a TensorFlow, PyTorch or ONNX AI model.
#
# Prerequisites: Python and the TensorFlow, PyTorch or ONNX library, generated thumbnails for images and videos, file hashes
#
# This feature assigns the following properties to the image and video files:
#     ai:csamDetector:label - classification, possible values are csam, porn or other
#     ai:csamDetector:csam, ai:csamDetector:porn, ai:csamDetector:other - confidence from 0 to 100 for each category
# For video files, the following properties are added as well:
#     ai:csamDetector:triggerFrame - frame number with the highest confidence from the video thumbnails
#     ai:csamDetector:hitPercentage - percentage of the total frames analyzed in the csam or porn categories
#     ai:csamDetector:avgConfidence - average of all the confidences of the chosen categorization
#
# To change the model to be used and other parameters, edit the file conf/CSAMDetector.txt
#
enableCSAMDetector = false

# Enables QR code detection and decoding.
enableQRCode = false

# Ignores and excludes duplicate files with the same hash from processing and from the case.
# WARNING: This feature is inherently DANGEROUS, as it could exclude duplicate files with
# different metadata (name, path, dates) from the case, such as email attachments or files in relevant folders. See #574 and #931.
ignoreDuplicates = false

# Exports the properties of all files in the case to the file "Lista de Arquivos.csv".
exportFileProps = true

# Processes file signatures.
processFileSignatures = true

# Enables parsing of file contents. It's necessary for several functions like
# container expansion, detection of encrypted content, metadata indexing, preview generation, etc.
enableFileParsing = true

# Expands containers, like compressed files, mailboxes, emails, and MS Office documents.
# File types to be expanded can be modified in the file "conf/CategoriesToExpand.txt".
expandContainers = true

# Processes supported disk images (like dd, e01, vhd, vmdk) contained in the main evidence recursively.
# Split images and snapshots are not supported for now.
processEmbeddedDisks = true

# Searches text extracted from items using regular expressions,
# such as social security/ID numbers, emails, URLs, credit cards, monetary values, etc.
# New expressions can be configured in the file "conf/RegexConfig.txt".
enableRegexSearch = true

# Enables automatic file exporting to the case folder based on categories or keywords.
# If enabled, only the exported files are included in the case, others are ignored, and the case becomes data-source independent.
# Relevant categories and keywords can be configured in CategoriesToExport.txt and KeywordsToExport.txt.
enableAutomaticExportFiles = false

# Enables detection of more than 70 languages in document files.
enableLanguageDetect = true

# Feature to recognize named entities: people, organizations, or places.
# StanfordCoreNLP 3.8 language models must be downloaded from https://stanfordnlp.github.io/CoreNLP/history.html and put in the plugins folder.
# This feature is CPU-heavy; it can increase processing time by 4x.
# Advanced settings can be modified in the "conf/NamedEntityRecognitionConfig.txt" file.
enableNamedEntityRecogniton = false

# Feature to create link graphs of communications (calls, messages, emails...).
# Optionally, it can create entities and links based on nearby regex hits (experimental). This can be enabled in conf/GraphConfig.json.
enableGraphGeneration = true

# Performs a test for randomness of binary items and unknown file types.
# This is used to detect possible encrypted items.
# Also makes indexing faster and reduces index size, especially when indexing unallocated space.
# May cause loss of hits surrounded by "random" content.
entropyTest = true

# Indexes file contents. If disabled, indexes only the properties of files.
indexFileContents = true

# Enables indexing to an ElasticSearch/OpenSearch cluster.
# You must configure at least the host/port options in conf/ElasticSearchConfig.txt.
enableIndexToElasticSearch = false

# Enables exporting files to a MinIO object storage cluster.
# You must configure at least the host/port options in conf/MinIOConfig.txt.
enableMinIO = false

# Enables OCR on images and scanned PDF files. It can increase processing time a lot.
# The OCR language dictionary and other advanced settings can be found in conf/OCRConfig.txt.
enableOCR = false

# Enables audio transcription.
# The default implementation uses VOSK transcription on the local CPU (faster, but poor accuracy).
# You can change the algorithm in conf/AudioTranscriptConfig.txt:
# - Wav2Vec2 algorithm (slower, with good accuracy)
# - Whisper algorithm (much slower, but better accuracy)
# - Google Cloud (about $1.00 per hour cost)
# - Microsoft Azure (about $1.00 per hour cost)
enableAudioTranscription = false

# Enables carving. "addUnallocated" must be enabled to scan unallocated space.
# By default, our carving module scans many places in the evidence for deleted or embedded files.
# Places to scan and file types to recover can be configured in "conf/CarverConfig.xml".
enableCarving = false

# Enables carving that retrieves known files from the LED database, based on the beginning (64K) of the file.
# It's necessary to enable "addUnallocated" and to configure "hashesDB" (with LED hash data imported).
enableLedCarving = false

# Enables specific carving of eMule "known.met" files. "addUnallocated" must be enabled.
enableKnownMetCarving = false

# Enables the generation of image thumbnails during processing.
# It can slow down processing, but greatly speeds up image preview in the gallery.
enableImageThumbs = true

# Enables the generation of image features during processing, allowing the search
# for visually similar images. "enableImageThumbs" must be enabled.
enableImageSimilarity = false

# Enables the face recognition feature.
# You must install Python and some dependencies; see https://github.com/sepinf-inc/IPED/wiki/User-Manual#facerecognition
# Advanced configuration options can be found in conf/FaceRecognitionConfig.txt.
# If enabled, you can search for faces from the analysis interface (check the options menu).
enableFaceRecognition = false

# Enables AI Summarization task with LLM models using a remote server.
# When enabled, summaries will be generated for supported files and displayed in the analysis interface.
# Configuration options can be found in conf/AISummarizationConfig.txt.
enableAISummarization = false

# Enables the age estimation feature.
# You must install Python and some dependencies; see https://github.com/sepinf-inc/IPED/wiki/User-Manual#AgeEstimation
# Advanced configuration options can be found in conf/AgeEstimationConfig.txt.
enableAgeEstimation = false

# Enables searching for crypto hardware wallets.
enableSearchHardwareWallets = true

# Enables extraction of image frames from video files.
# Extraction settings can be modified in the file "conf/VideoThumbsConfig.txt".
enableVideoThumbs = true

# Enables thumbnail creation for PDFs and many formats supported by LibreOffice.
# Advanced settings can be found in conf/DocThumbsConfig.txt.
enableDocThumbs = false

# Enables HTML report generation on automatic extractions or from selected items.
# Generation settings can be modified in the file "conf/HTMLReportConfig.txt".
enableHTMLReport = true

# Enables the remote image classifier.
# It performs classification of image and video files.
# Advanced settings can be modified in the file "conf/RemoteImageClassifierConfig.txt".
enableRemoteImageClassifier = false
